function multiplyAsync(a, b) { return new Promise((resolve, reject) => { if (typeof a !== "number" || typeof b !== "number") { reject("Некоректні значення"); return; } setTimeout(() => { resolve(a * b); }, 2000); }); } function sortArrayAsync(array) { return new Promise((resolve, reject) => { if (array.length === 0) { reject("Масив порожній"); return; } setTimeout(() => { resolve(array.sort((a, b) => a - b)); }, 2000); }); } async function main() { try { const result1 = await multiplyAsync(6, 9); console.log(result1); const result2 = await sortArrayAsync([5, 2, 8, 1, 3]); console.log(result2); } catch (error) { console.log(error); } } main();
